SELECT
    r.expected_role,
    'api',
    NOW(),
    NOW(),
    r.business_id,
    1,
    0,
    0,
    NULL
FROM (
    -- Generate expected roles for each existing business_id (excluding NULL)
    SELECT DISTINCT business_id, CONCAT(role_type, '#', business_id) AS expected_role
    FROM (
        SELECT business_id, 'business_admin' AS role_type FROM roles WHERE business_id IS NOT NULL
        UNION ALL
        SELECT business_id, 'business_staff' FROM roles WHERE business_id IS NOT NULL
        UNION ALL
        SELECT business_id, 'business_administrator' FROM roles WHERE business_id IS NOT NULL
        UNION ALL
        SELECT business_id, 'business_teacher' FROM roles WHERE business_id IS NOT NULL
        UNION ALL
        SELECT business_id, 'agency' FROM roles WHERE business_id IS NOT NULL
    ) AS expected_roles
) AS r
LEFT JOIN roles real_roles
    ON real_roles.name = r.expected_role AND real_roles.business_id = r.business_id
WHERE real_roles.id IS NULL;
